87  void SetPipCount(uint32_t pipCount)
88  {
89  m_pipStripDataProperties.m_numberOfPips = pipCount;
90 
91  float totalSpacing = m_pipStripDataProperties.m_pipSpacing * (int(pipCount) - 1);
92 
93  Vector2 sizeOfAllPips;
94  switch (m_pipStripDataProperties.m_stripOrientation)
95  {
96  case PipStripOrientation::LeftToRight:
97  sizeOfAllPips = DirectX::SimpleMath::Vector2(
98  m_pipStripDataProperties.m_pipSize.x * pipCount + totalSpacing,
99  m_pipStripDataProperties.m_pipSize.y );
100  break;
101  case PipStripOrientation::TopToBottom:
102  sizeOfAllPips = DirectX::SimpleMath::Vector2(
103  m_pipStripDataProperties.m_pipSize.x,
104  m_pipStripDataProperties.m_pipSize.y * pipCount + totalSpacing);
105  break;
106  default:
107  throw std::exception("Unhandled pip strip orientation in method UIPipStrip::SetPipCount()");
108  }
109 
110  UIElement::SetRelativeSizeInRefUnits(sizeOfAllPips);
