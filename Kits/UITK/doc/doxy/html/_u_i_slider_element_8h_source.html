49  float CalculateValue(float t) const
50  {
51  // support discrete values
52  if (sliderType == SliderType::Discrete)
53  {
54  auto floatDiscreteSteps = float(numDiscreteSteps);
55  t = float(int(0.5f + (t * floatDiscreteSteps))) / floatDiscreteSteps;
56  }
57  return valueRange.x + (t * (valueRange.y - valueRange.x));
58  }
59 
61  float CalculateInterpolant(float value) const
62  {
63  if (value <= valueRange.x)
64  {
65  return 0.0f;
66  }
67  else if (value >= valueRange.y)
68  {
69  return 1.0f;
70  }
71  else
72  {
73  return (value - valueRange.x) / (valueRange.y - valueRange.x);
74  }
75  }
